166 Error Read(uint8_t reg, std::span<uint8_t> out) noexcept {
167 if (out.empty() || (out.size() > kMaxBurst)) {
168 return Error::InvalidArgument;
169 }
170 // The buffers the port sees are members of this object, never on the caller's stack, so a
171 // port whose bus reads them by DMA reaches them (hardware lessons 20).
172 pointer_[0] = reg;
173 if (!port_.WriteRead(address_, pointer_.data(), pointer_.size(), rx_.data(), out.size())) {
174 return Error::Bus;
175 }
176 for (std::size_t i = 0U; i < out.size(); ++i) {
177 out[i] = rx_[i];
178 }
179 return Error::Ok;
180 }
181
183 template <std::size_t N>
184 requires BurstSize<N>
185 Error Write(uint8_t reg, std::span<const uint8_t, N> data) noexcept {
186 return Write(reg, std::span<const uint8_t>{data});
187 }
188
191 Error Write(uint8_t reg, std::span<const uint8_t> data) noexcept {
192 if (data.empty() || (data.size() > kMaxBurst)) {
193 return Error::InvalidArgument;
194 }
195 frame_[0] = reg;
196 const std::size_t count = data.size();
197 for (std::size_t i = 0U; i < count; ++i) {
198 frame_[1U + i] = data[i];
199 }
200 return port_.Write(address_, frame_.data(), count + 1U) ? Error::Ok : Error::Bus;
201 }
202
203private:
204 Port& port_;
205 uint8_t address_;
206 std::array<uint8_t, 1> pointer_{};
207 std::array<uint8_t, kMaxBurst> rx_{};
208 std::array<uint8_t, 1U + kMaxBurst> frame_{};
209};
