98constexpr Result<StartupProfile> StartupProfileOf(Option option) noexcept {
99 const auto n = static_cast<std::size_t>(detail::ToUnderlying(option));
100 if ((n < 1U) || (n > detail::kPowerUpOrder.size())) {
101 return Result<StartupProfile>::Failure(Error::InvalidArgument);
102 }
103 StartupProfile p{};
104 p.order = detail::kPowerUpOrder[n - 1U];
105 p.slot_us = (option == Option::A1) ? kSlotFastUs : kSlotSlowUs;
106 p.pwron_edge = (option == Option::A8);
107 p.sw1_dvs = detail::kSw1Dvs[n - 1U];
108 p.sw2_dvs = detail::kSw2Dvs[n - 1U];
109 return Result<StartupProfile>::Success(p);
110}

118constexpr uint32_t PowerUpDelayUs(const StartupProfile& profile, Rail rail) noexcept {
119 const uint32_t order =
120 profile.order[static_cast<std::size_t>(detail::ToUnderlying(rail)) % kRailCount];
121 return (order > 1U) ? ((order - 1U) * profile.slot_us) : 0U;
122}

129constexpr uint32_t PowerUpDurationUs(const StartupProfile& profile) noexcept {
130 uint32_t longest = 0U;
131 for (const uint8_t order : profile.order) {
132 const uint32_t d = (order > 1U) ? ((static_cast<uint32_t>(order) - 1U) * profile.slot_us) : 0U;
133 longest = (d > longest) ? d : longest;
134 }
135 return longest;
136}

159constexpr Result<uint32_t> ColdStartToResetReleaseUs(const StartupProfile& profile,
160 uint32_t por_dly_code) noexcept {
161 const Result<uint32_t> ms = ResetReleaseDelayMs(por_dly_code);
162 if (!ms.ok()) {
163 return Result<uint32_t>::Failure(Error::InvalidArgument);
164 }
165 return Result<uint32_t>::Success(kVsnvsTurnOnDelayUs + PowerUpDurationUs(profile) +
166 (ms.value * 1000U));
167}

174constexpr bool IsPowerDownBypassed(const PowerDownSequence& sequence) noexcept {
175 bool bypassed = true;
176 for (const uint8_t order : sequence.order) {
177 bypassed = bypassed && (order == 0U);

179 return bypassed;

190constexpr uint32_t PowerDownDelayUs(const PowerDownSequence& sequence, Rail rail) noexcept {
191 uint32_t highest = 0U;
192 for (const uint8_t order : sequence.order) {
193 highest = (order > highest) ? order : highest;
194 }
195 const uint32_t own =
196 sequence.order[static_cast<std::size_t>(detail::ToUnderlying(rail)) % kRailCount];
197 return (highest - own) * kPowerDownStepUs;
198}

205constexpr uint32_t PowerDownDurationUs(const PowerDownSequence& sequence) noexcept {
206 uint32_t highest = 0U;
207 for (const uint8_t order : sequence.order) {
208 highest = (order > highest) ? order : highest;
209 }
210 return highest * kPowerDownStepUs;
