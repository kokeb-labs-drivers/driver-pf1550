285constexpr Error ValidateLimits(const Limits& l) noexcept {
286 if (!InRange(l.cell_max_mv, kChargeVoltageMinMv, kChargeVoltageMaxMv) ||
287 !InRange(l.cell_max_charge_ma, kChargeCurrentMinMa, kChargeCurrentMaxMa) ||
288 !InRange(l.input_max_ma, 10_u32, 1500_u32)) {
289 return Error::InvalidArgument;
290 }
291 for (const RailLimits& r : l.rail) {
292 if (r.min_mv > r.max_mv) {
293 return Error::InvalidArgument;
294 }
295 }
296 return Error::Ok;
297}

300constexpr Error ValidateChargerValues(const ChargerConfig& c, const Limits& l) noexcept {
301 if (!ChargeVoltageCode(c.charge_voltage_mv).ok() ||

381constexpr Error ValidateModes(const RailModes& m, const RailLimits& r,
382 const Acknowledgements& a) noexcept {

397constexpr Error ValidateVolt(Rail rail, Microvolts v, const Limits& l) noexcept {
398 if (v.value == 0U) {
399 return Error::Ok;
400 }
401 const bool buck = (rail == Rail::Sw1) || (rail == Rail::Sw2);
402 const bool on_grid = buck ? BuckDvsCode(v.value).ok() : LdoCode(rail, v.value).ok();
403 if (!on_grid) {
404 return Error::InvalidArgument;
405 }
406 const RailLimits& r = LimitsOf(l, rail);
407 const uint32_t mv = v.value / 1000U;
408 return ((r.max_mv != 0U) && (mv >= r.min_mv) && (mv <= r.max_mv)) ? Error::Ok
409 : Error::OutOfLimits;
410}

413constexpr Error ValidateBuck(const BuckConfig& b, Rail rail, const Config& c) noexcept {
414 if (!b.apply) {
415 return Error::Ok;
416 }
417 if (!IsEnumerator(b.dvs) || !IsEnumerator(b.current_limit) || !IsEnumerator(b.dvs_speed)) {
418 return Error::InvalidArgument;
419 }
420 const bool sets_volt =
421 (b.run_uv.value != 0U) || (b.standby_uv.value != 0U) || (b.sleep_uv.value != 0U);
422 if (sets_volt && ((rail == Rail::Sw3) || (b.dvs != Dvs::Enabled))) {
423 return Error::ReadOnlyRail;
424 }
425 for (const Microvolts v : {b.run_uv, b.standby_uv, b.sleep_uv}) {
426 if (const Error e = ValidateVolt(rail, v, c.limits); e != Error::Ok) {
427 return e;
428 }
429 }
430 return ValidateModes(b.modes, LimitsOf(c.limits, rail), c.ack);
431}

445constexpr Error ValidateRails(const Config& c) noexcept {
446 const std::array<Error, 3> buck = {{ValidateBuck(c.sw1, Rail::Sw1, c),
447 ValidateBuck(c.sw2, Rail::Sw2, c),
448 ValidateBuck(c.sw3, Rail::Sw3, c)}};
449 const std::array<Error, 3> ldo = {{ValidateLdo(c.ldo1, Rail::Ldo1, c),
450 ValidateLdo(c.ldo2, Rail::Ldo2, c),
451 ValidateLdo(c.ldo3, Rail::Ldo3, c)}};
452 for (const Error e : buck) {
453 if (e != Error::Ok) {
454 return e;
455 }
456 }
457 for (const Error e : ldo) {
458 if (e != Error::Ok) {
459 return e;
460 }
461 }
462 if (c.vrefddr.apply) {
463 return ValidateModes(c.vrefddr.modes, LimitsOf(c.limits, Rail::VrefDdr), c.ack);
464 }
465 return Error::Ok;

469constexpr Error ValidatePeripherals(const Config& c) noexcept {
470 if (c.coin.apply) {
471 if (!CoinVoltageCode(c.coin.voltage_mv).ok()) {
472 return Error::InvalidArgument;
473 }
474 if (c.coin.charge_enable && !c.ack.coin_cell_is_rechargeable) {
475 return Error::Unsafe;
476 }
477 if ((c.coin.voltage_mv > 1800U) && (c.limits.coin_cell_max_mv < 3300U)) {
478 return Error::OutOfLimits;
479 }
480 }
481 if (c.usbphy.apply && !IsEnumerator(c.usbphy.voltage)) {
482 return Error::InvalidArgument;
483 }
484 if (c.led.apply &&
485 ((c.led.duty_32nds > 32U) || !IsEnumerator(c.led.frequency) || !IsEnumerator(c.led.mode))) {
486 return Error::InvalidArgument;
487 }
488 return Error::Ok;

492constexpr Error ValidatePowerControl(const PowerConfig& p, const Acknowledgements& a) noexcept {
493 if (!p.apply) {
494 return Error::Ok;
495 }
496 const bool enums = AllEnumerators(p.standby_delay, p.pwron_debounce, p.onkey_debounce,
497 p.low_sys_warn, p.global_reset, p.uvdet);
498 if (!enums) {
499 return Error::InvalidArgument;
500 }
501 const bool unsafe = Unacknowledged(p.write_reset_paths, a.hard_reset_paths) ||
502 Unacknowledged(p.write_uvdet, a.undervoltage_threshold);
503 return unsafe ? Error::Unsafe : Error::Ok;

507constexpr Error ValidatePowerDown(const PowerDownConfig& d, const Acknowledgements& a) noexcept {

528constexpr Error ValidateOrder(const Config& c) noexcept {
529 std::array<bool, kRailCount> seen{};
530 for (const Rail r : c.rail_order) {
531 if (!IsEnumerator(r)) {
532 return Error::InvalidArgument;

534 seen[static_cast<std::size_t>(ToUnderlying(r))] = true;
