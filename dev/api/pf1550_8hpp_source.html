162[[nodiscard]] constexpr Fault ClassifyFault(ChargerState state, BatteryState battery, bool watchdog,
163 bool battery_oc) noexcept {
164 if (state == ChargerState::ThermalShutdown) {
165 return Fault::ThermalShutdown;
166 }
167 if (state == ChargerState::TimerFault) {
168 return Fault::TimerFault;
169 }
170 if ((state == ChargerState::BatteryOvervoltage) || (battery == BatteryState::Overvoltage)) {
171 return Fault::BatteryOvervoltage;
172 }
173 if (battery_oc) {
174 return Fault::BatteryOvercurrent;
175 }
176 if (watchdog) {
177 return Fault::WatchdogExpired;
178 }
179 if (battery == BatteryState::NotDetected) {
180 return Fault::BatteryMissing;
181 }
182 if (state == ChargerState::ThermistorSuspend) {
183 return Fault::ThermistorSuspend;
184 }
185 if ((state == ChargerState::Unknown) || (battery == BatteryState::Unknown)) {
186 return Fault::UnknownState;
187 }
188 return Fault::None;
189}

192[[nodiscard]] constexpr VbusFlags DecodeVbus(uint8_t raw) noexcept {
193 const auto bits = static_cast<uint32_t>(raw);
194 VbusFlags f{};
195 f.valid = (bits & reg::kVbusValid) != 0U;
196 f.ovlo = (bits & reg::kVbusOvlo) != 0U;
197 f.uvlo = (bits & reg::kVbusUvlo) != 0U;
198 f.in2sys = (bits & reg::kVbusIn2Sys) != 0U;
199 f.dpm = (bits & reg::kVbusDpm) != 0U;
200 return f;
201}
