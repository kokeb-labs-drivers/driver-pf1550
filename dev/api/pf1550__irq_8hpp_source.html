84constexpr uint8_t CategoryBitsOf(IrqGroup group) noexcept {

120constexpr bool Has(const InterruptEvents& events, IrqSource source) noexcept {
121 const IrqLocation loc = LocationOf(source);
122 const uint32_t byte =
123 events.status[static_cast<std::size_t>(detail::ToUnderlying(loc.group)) % kIrqGroups];
124 return ((byte >> (loc.bit & 7U)) & 1U) != 0U;
125}

131constexpr bool IsFaultSource(IrqSource source) noexcept {
132 // SW1..3 low-side (7..9) and high-side (10..12) limits, LDO faults (15..17), 125 C (19), SYS_OVLO
133 // (30).
134 constexpr uint32_t kFaultSources = (0x3FU << 7) | (0x7U << 15) | (1U << 19) | (1U << 30);
135 const auto i = static_cast<uint32_t>(detail::ToUnderlying(source));
136 return (i < kIrqSourceCount) && (((kFaultSources >> i) & 1U) != 0U);
137}

142constexpr bool HasFault(const std::array<uint8_t, kIrqGroups>& status) noexcept {
143 InterruptEvents e{};
144 e.status = status;
145 for (std::size_t i = 0U; i < kIrqSourceCount; ++i) {
146 const auto source = static_cast<IrqSource>(i);
147 if (IsFaultSource(source) && Has(e, source)) {
148 return true;
149 }
150 }
151 return false;
152}

165template <std::size_t N>
166constexpr IrqMask UnmaskOnly(const std::array<IrqSource, N>& sources) noexcept {
167 IrqMask m{};
168 for (const IrqSource s : sources) {
169 const IrqLocation loc = LocationOf(s);
170 auto& byte = m.masked[static_cast<std::size_t>(detail::ToUnderlying(loc.group)) % kIrqGroups];
171 byte = static_cast<uint8_t>(static_cast<uint32_t>(byte) & ~(1U << (loc.bit & 7U)));
172 }
173 return m;
174}
