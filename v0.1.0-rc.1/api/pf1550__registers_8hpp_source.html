368constexpr Access AccessOf(uint8_t address) noexcept {
369 for (const RegisterInfo& info : kRegisters) {
370 if (info.address == address) {
371 return info.access;
372 }
373 }
374 return Access::Never;
375}
376
382constexpr bool IsDocumented(uint8_t address) noexcept {
383 bool found = false;
384 for (const RegisterInfo& info : kRegisters) {
385 found = found || (info.address == address);
386 }
387 return found;
388}
389
396constexpr bool IsWritable(uint8_t address) noexcept {
397 const Access access = AccessOf(address);
398 return (access != Access::ReadOnly) && (access != Access::Never);
399}
