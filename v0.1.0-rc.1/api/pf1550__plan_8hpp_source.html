105constexpr uint8_t SelfChangingBits(uint8_t address) noexcept {
106 if (address == reg::kChgOper) {
107 return reg::ChgOperMode::kMask;
108 }
109 if (address == reg::kBatfetCnfg) {
110 return reg::Wdtclr::kMask;
111 }
112 for (const RailRegs& r : kRailRegs) {
113 if ((r.ctrl == address) && (address != 0U)) {
114 return reg::kCtrlEn;
115 }
116 }
117 return 0U;

167constexpr RegWrite BuckCtrlWrite(Rail rail, const BuckConfig& b) noexcept {
168 const bool dvs_rail = (rail != Rail::Sw3);
169 uint8_t mask = Or(kModeMask, Or(reg::kBuckFpwm, Or(reg::kBuckFpwmInDvs, reg::kBuckRdisEnb)));
170 uint8_t value = Or(ModeBits(b.modes), Or(BitIf(b.fpwm, reg::kBuckFpwm),
171 Or(BitIf(b.fpwm_in_dvs, reg::kBuckFpwmInDvs),
172 BitIf(!b.output_discharge, reg::kBuckRdisEnb))));
173 if (!dvs_rail) {
174 // SW3_CTRL bits 4 and 5 are UNUSED (Table 125): neither is written.
175 mask = static_cast<uint8_t>(static_cast<uint32_t>(mask) &
176 ~static_cast<uint32_t>(reg::kBuckFpwmInDvs));
177 value = static_cast<uint8_t>(static_cast<uint32_t>(value) & static_cast<uint32_t>(mask));
178 }
179 if (dvs_rail) {
180 mask = Or(mask, reg::kBuckDvsSpeed);
181 value = Or(value, BitIf(b.dvs_speed == DvsSpeed::Us4, reg::kBuckDvsSpeed));
182 }
183 return MakeWrite(RegsOf(rail).ctrl, mask, value);
184}

196constexpr std::array<RegWrite, 5> BuckPlan(Rail rail, const BuckConfig& b) noexcept {
197 const RailRegs& r = RegsOf(rail);
198 const bool volts = (b.dvs == Dvs::Enabled) && (rail != Rail::Sw3);
199 return {
200 {volts ? BuckVoltWrite(r.volt, b.run_uv) : RegWrite{},
201 volts ? BuckVoltWrite(r.stby_volt, b.standby_uv) : RegWrite{},
202 volts ? BuckVoltWrite(r.slp_volt, b.sleep_uv) : RegWrite{}, BuckCtrlWrite(rail, b),
203 MakeWrite(r.ctrl1, reg::BuckIlim::kMask, static_cast<uint8_t>(CodeOf(b.current_limit)))}};
204}

224constexpr std::array<RegWrite, 5> PeripheralPlan(const Config& c) noexcept {
225 RegWrite vsnvs{};
226 RegWrite coin{};
227 RegWrite phy{};
228 RegWrite led_pwm{};
229 RegWrite led_cfg{};
230 if (c.vsnvs.apply) {
231 vsnvs = MakeWrite(reg::kVsnvsCtrl, reg::kVsnvsLibgdis,
232 BitIf(c.vsnvs.local_bandgap_disable, reg::kVsnvsLibgdis));
233 }
234 if (c.coin.apply) {
235 const uint8_t code = CoinVoltageCode(c.coin.voltage_mv).value;
236 coin = MakeWrite(reg::kCoinCellControl, Or(reg::CoinVcoin::kMask, reg::kCoinChen),
237 Or(code, BitIf(c.coin.charge_enable, reg::kCoinChen)));
238 }
239 if (c.usbphy.apply) {
240 phy = MakeWrite(reg::kUsbPhyLdoCnfg, Or(reg::kUsbPhyLdo, Or(reg::kUsbPhy49, reg::kActDisPhy)),
241 Or(BitIf(c.usbphy.enable, reg::kUsbPhyLdo),
242 Or(BitIf(c.usbphy.voltage == UsbPhyVoltage::V4p9, reg::kUsbPhy49),
243 BitIf(c.usbphy.active_discharge, reg::kActDisPhy))));
244 }
245 if (c.led.apply) {
246 const LedConfig& l = c.led;
247 led_pwm = MakeWrite(reg::kLedPwm, Or(reg::LedPwmDuty::kMask, Or(reg::kLedRamp, reg::kLedEn)),
248 Or(static_cast<uint8_t>(l.duty_32nds),
249 Or(BitIf(l.ramp, reg::kLedRamp), BitIf(l.enable, reg::kLedEn))));
250 led_cfg = MakeWrite(
251 reg::kLedCnfg,
252 Or(reg::LedFreq::kMask, Or(reg::LedCurrent::kMask, Or(reg::kLedCfg, reg::kLedOvrd))),
253 Or(static_cast<uint8_t>(CodeOf(l.frequency)),
254 Or(reg::kLedCurrent6mA, Or(BitIf(l.mode == LedMode::FlashWhileCharging, reg::kLedCfg),
255 BitIf(l.software_override, reg::kLedOvrd)))));
256 }
257 return {{vsnvs, coin, phy, led_pwm, led_cfg}};
258}

261constexpr std::array<RegWrite, 3> PowerPlan(const PowerConfig& p) noexcept {
262 if (!p.apply) {
263 return {};
264 }
265 // Each unguarded field is written only when its own flag is set: a default must not override a
266 // value the board chose, and STANDBYINV flips the state with the pin at a steady level.
267 uint8_t m0 = p.write_standby ? Or(reg::StandbyDly::kMask, reg::kStandbyInv) : 0_u8;
268 uint8_t v0 = p.write_standby ? Or(static_cast<uint8_t>(CodeOf(p.standby_delay)),
269 BitIf(p.standby_inverted, reg::kStandbyInv))
270 : 0_u8;
271 uint8_t m1 = p.write_debounce ? Or(reg::PwronDbnc::kMask, reg::OnkeyDbnc::kMask) : 0_u8;
272 uint8_t v1 = p.write_debounce ? Or(static_cast<uint8_t>(DebounceCode(p.pwron_debounce)),
273 static_cast<uint8_t>(DebounceCode(p.onkey_debounce) << 2))
274 : 0_u8;
275 if (p.write_regscp) {
276 m1 = Or(m1, reg::kRegScpEn);
277 v1 = Or(v1, BitIf(p.ldo_off_on_current_limit, reg::kRegScpEn));
278 }
279 uint8_t m2 = p.write_low_sys_warn ? reg::LowSysWarn::kMask : 0_u8;
280 auto v2 = p.write_low_sys_warn ? static_cast<uint8_t>(CodeOf(p.low_sys_warn) << 2) : 0_u8;
281 if (p.write_reset_paths) {
282 m0 = Or(m0, reg::TgReset::kMask);
283 v0 = Or(v0, static_cast<uint8_t>(CodeOf(p.global_reset) << 6));
284 m1 = Or(m1, Or(reg::kPwronRstEn, Or(reg::kRestartEn, reg::kOnkeyRstEn)));
285 v1 = Or(v1, Or(BitIf(p.pwron_long_press_reset, reg::kPwronRstEn),
286 Or(BitIf(p.restart_on_long_press, reg::kRestartEn),
287 BitIf(p.onkey_reset, reg::kOnkeyRstEn))));
288 }
289 if (p.write_uvdet) {
290 m2 = Or(m2, reg::Uvdet::kMask);
291 v2 = Or(v2, static_cast<uint8_t>(CodeOf(p.uvdet)));
292 }
293 return {{MakeWrite(reg::kPwrCtrl0, m0, v0), MakeWrite(reg::kPwrCtrl1, m1, v1),
294 MakeWrite(reg::kPwrCtrl2, m2, v2)}};
295}

298constexpr std::array<RegWrite, kRailCount> PowerDownPlan(const PowerDownSequence& s) noexcept {
299 std::array<RegWrite, kRailCount> plan{};
300 for (std::size_t i = 0U; i < kRailCount; ++i) {
301 plan[i] =
302 MakeWrite(kPwrdnSeqRegs[i], reg::PwrdnSeq::kMask,
303 static_cast<uint8_t>(static_cast<uint32_t>(s.order[i]) & reg::PwrdnSeq::kMask));
304 }
305 return plan;
306}

310constexpr std::array<RegWrite, 12> ChargerPlan(const ChargerConfig& c) noexcept {
311 const BatfetConfig& b = c.batfet;
312 const uint8_t cc = ChargeCurrentCode(c.charge_current_ma).value;
313 const uint8_t cv = ChargeVoltageCode(c.charge_voltage_mv).value;
314 const uint32_t ieoc = TerminationCode(c.termination_ma).value;
315 const uint32_t ilim = InputLimitCode(c.input_limit_ma).value;
316 const uint8_t tmr = Or(static_cast<uint8_t>(CodeOf(c.fast_timer)),
317 Or(static_cast<uint8_t>(CodeOf(c.eoc_timer) << 3),
318 BitIf(c.precharge_timer == PrechargeTimer::Min45, reg::kTprechg)));
319 const uint8_t tmr_mask = Or(reg::Fchgtime::kMask, Or(reg::Eoctime::kMask, reg::kTprechg));
320 const uint8_t fet_mask =
321 Or(reg::BatfetOc::kMask, Or(reg::kBovrcDisbatfet, Or(reg::kWdTime, reg::kBovrcNovbus)));
322 const uint8_t fet = Or(static_cast<uint8_t>(CodeOf(b.oc_threshold) << 4),
323 Or(BitIf(b.oc_keeps_closed, reg::kBovrcDisbatfet),
324 Or(BitIf(c.watchdog.time == WatchdogTime::S32, reg::kWdTime),
325 BitIf(b.oc_without_vbus, reg::kBovrcNovbus))));
326 const uint8_t thm_mask =
327 Or(reg::ThmCnfg::kMask,
328 Or(reg::Regtemp::kMask, Or(reg::kThmCold, Or(reg::kThmHot, reg::kTempFbEn))));
329 const uint8_t thm = Or(static_cast<uint8_t>(CodeOf(c.thermistor)),
330 Or(static_cast<uint8_t>(CodeOf(c.thermal_reg) << 2),
331 Or(BitIf(c.jeita.cold == Cold::CMinus10, reg::kThmCold),
332 Or(BitIf(c.jeita.hot == Hot::C55, reg::kThmHot),
333 BitIf(c.thermal_foldback, reg::kTempFbEn)))));
334 const uint8_t adj_mask =
335 Or(reg::kThmWarm, Or(reg::kThmCool, Or(reg::CvAdj::kMask, reg::CcAdj::kMask)));
336 const uint8_t adj = Or(Or(BitIf(c.jeita.warm == Warm::C50, reg::kThmWarm),
337 BitIf(c.jeita.cool == Cool::C10, reg::kThmCool)),
338 Or(static_cast<uint8_t>(CodeOf(c.jeita.cv_adjust) << 2),
339 static_cast<uint8_t>(CodeOf(c.jeita.cc_adjust) << 4)));
340 const VbusOptions& v = c.vbus;
341 return {
342 {MakeWrite(reg::kChgTmr, tmr_mask, tmr),
343 MakeWrite(reg::kChgEocCnfg, Or(reg::ChgRestart::kMask, reg::Ieoc::kMask),
344 Or(static_cast<uint8_t>(CodeOf(c.restart)), static_cast<uint8_t>(ieoc << 4U))),
345 MakeWrite(reg::kChgCurrCnfg, reg::ChgCc::kMask, cc),
346 MakeWrite(reg::kBattReg, Or(reg::Chgcv::kMask, reg::VsysMin::kMask),
347 Or(cv, static_cast<uint8_t>(CodeOf(c.vsys_min) << 6))),
348 MakeWrite(reg::kBatfetCnfg, fet_mask, fet), MakeWrite(reg::kThmRegCnfg, thm_mask, thm),
349 MakeWrite(reg::kVbusInlimCnfg, reg::VbusLinIlim::kMask, static_cast<uint8_t>(ilim << 3U)),
350 MakeWrite(reg::kVbusLinDpm, Or(reg::VbusDpmReg::kMask, reg::kVinDpmStop),
351 Or(static_cast<uint8_t>(CodeOf(v.dpm)),
352 BitIf(v.dpm_stop == DpmStop::Mv250, reg::kVinDpmStop))),
353 MakeWrite(reg::kDbncDelayTime, Or(reg::VbusOvTdb::kMask, reg::SysWkupDly::kMask),
354 Or(static_cast<uint8_t>(ShortDebounceCode(v.ovlo_debounce)),
355 static_cast<uint8_t>(CodeOf(v.sys_wakeup_delay) << 4))),
356 MakeWrite(reg::kChgIntCnfg, Or(reg::kChgIntGen, reg::kEocInt),
357 Or(BitIf(c.interrupts.every_transition, reg::kChgIntGen),
358 BitIf(c.interrupts.eoc, reg::kEocInt))),
359 MakeWrite(reg::kThmAdjSetting, adj_mask, adj),
360 MakeWrite(
361 reg::kVbus2SysCnfg, Or(reg::Vbus2SysTdb::kMask, reg::kVbus2SysThrsh),
362 Or(static_cast<uint8_t>(ShortDebounceCode(v.vbus2sys_debounce)),
363 BitIf(v.vbus2sys_threshold == Vbus2SysThreshold::Mv175, reg::kVbus2SysThrsh)))}};
364}
