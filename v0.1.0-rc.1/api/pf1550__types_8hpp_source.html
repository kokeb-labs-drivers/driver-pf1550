68constexpr const char* ToString(Error error) noexcept {

70 case Error::Ok:

72 case Error::NotInitialized:
73 return "NotInitialized";
74 case Error::InvalidArgument:
75 return "InvalidArgument";
76 case Error::Bus:
77 return "Bus";
78 case Error::Timeout:
79 return "Timeout";
80 case Error::DeviceIdMismatch:
81 return "DeviceIdMismatch";
82 case Error::Crc:
83 return "Crc";
84 case Error::Busy:
85 return "Busy";
86 case Error::DeviceFault:
87 return "DeviceFault";
88 case Error::Unsupported:
89 return "Unsupported";
90 case Error::OutOfLimits:
91 return "OutOfLimits";
92 case Error::ReadOnlyRail:
93 return "ReadOnlyRail";
94 case Error::Unsafe:
95 return "Unsafe";
96 case Error::Refused:
97 return "Refused";
98 case Error::ChargerFault:
99 return "ChargerFault";
100 case Error::VerifyFailed:
101 return "VerifyFailed";
102 case Error::NoDevice:
103 return "NoDevice";
104 case Error::DeviceReset:
105 return "DeviceReset";
