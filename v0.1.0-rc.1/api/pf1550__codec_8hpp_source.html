59constexpr Result<uint8_t> ChargeVoltageCode(uint32_t mv) noexcept {

206constexpr Result<uint8_t> LdoCode(Rail rail, uint32_t uv) noexcept {
207 if (rail == Rail::Ldo2) {
208 if ((uv >= 1800000U) && (uv <= 3300000U) && (((uv - 1800000U) % 100000U) == 0U)) {
209 return Result<uint8_t>::Success(static_cast<uint8_t>((uv - 1800000U) / 100000U));
210 }
211 return Result<uint8_t>::Failure(Error::InvalidArgument);
212 }
213 if ((rail != Rail::Ldo1) && (rail != Rail::Ldo3)) {
214 return Result<uint8_t>::Failure(Error::InvalidArgument);
215 }
216 if ((uv >= 750000U) && (uv <= 1500000U) && (((uv - 750000U) % 50000U) == 0U)) {
217 return Result<uint8_t>::Success(static_cast<uint8_t>((uv - 750000U) / 50000U));
218 }
219 if ((uv >= 1800000U) && (uv <= 3300000U) && (((uv - 1800000U) % 100000U) == 0U)) {
220 return Result<uint8_t>::Success(static_cast<uint8_t>(16U + ((uv - 1800000U) / 100000U)));
221 }
222 return Result<uint8_t>::Failure(Error::InvalidArgument);
223}

231constexpr uint32_t LdoUv(Rail rail, uint32_t code) noexcept {
232 if (rail == Rail::Ldo2) {
233 return 1800000U + (100000U * (code & 0x0FU));
234 }
235 if ((rail != Rail::Ldo1) && (rail != Rail::Ldo3)) {
236 return 0U;
237 }
238 const uint32_t c = code & 0x1FU;
239 return (c < 16U) ? (750000U + (50000U * c)) : (1800000U + (100000U * (c - 16U)));
240}

259constexpr PowerState DecodePowerState(uint32_t code) noexcept {
260 switch (code) {
261 case 0x00U:
262 return PowerState::Wait;
263 case 0x0CU:
264 return PowerState::Run;
265 case 0x0DU:
266 return PowerState::Standby;
267 case 0x0EU:
268 return PowerState::Sleep;
269 case 0x2BU:
270 return PowerState::RegsDisable;
271 default:
272 break;
273 }
274 return PowerState::Unknown;
275}
